   384→   384→  1348→    (config.englishLabel ? (9 * 23 + 500) : 0) +

   386→   386→  1350→    (config.subtitle ? (12 * 23 + 200) : 0);

   481→   481→  1445→### Recipe R5: Clean White (Academic)
   482→   482→  1446→
   483→   483→  1447→**Visual:** Pure white background + school name + centered title + 2-column meta info table with underlines + footer
   484→   484→  1448→**Use case:** academic thesis, standards documents
   485→   485→  1449→**Architecture:** 16838 outer wrapper (white fill, invisible) + cell margins for page margin simulation. No top/bottom decorative lines.
   486→   486→  1450→
   487→   487→  1451→**Meta info table rules (cross-engine safe):**
   488→   488→  1452→- 2-column table: label + value, **percentage widths only** (`WidthType.PERCENTAGE`)
   489→   489→  1453→- **Table width is adaptive:** 55–75% of page, calculated by `calcR5MetaLayout()`. Table centered via `alignment: CENTER`.
   490→   490→  1454→- **Label column:** adaptive 25–45% of table, **LEFT aligned**, plain text with "：" appended. NO full-width space padding, NO right-alignment.
   491→   491→  1455→- **Label column borders:** none (no bottom border on label cells).
   492→   492→  1456→- **Value column:** remaining %, **LEFT aligned**, `bottom: single sz=4` border = fixed-length underline (consistent length for all rows regardless of text).
   493→   493→  1457→- No left/right/top borders on either column.
   494→   494→  1458→- ⚠️ Do NOT use DXA widths, full-width space padding (`\u3000`), spacer columns, or tab stops — WPS renders them inconsistently.
   495→   495→  1459→- ⚠️ Do NOT use `margins.top` on the wrapper cell — use `spacing.before` on first paragraph instead.
   496→   496→  1460→
   497→   497→  1461→**Known limitation:** When meta lines ≥ 6 AND title has 3 lines, MS Office may render content slightly taller than WPS, potentially clipping the footer line. Mitigate by reducing `midSpacing` or using a smaller title font.
   498→   498→  1462→
   499→   499→  1463→```js
   500→   500→  1464→// ── Width-aware title layout (handles mixed Chinese + English) ──
   501→   501→  1465→function estimateTextWidth(text, pt) {
   502→   502→  1466→  let width = 0;
   503→   503→  1467→  for (const ch of text) {
   504→   504→  1468→    const code = ch.codePointAt(0);
   505→   505→  1469→    const isCJK = (code >= 0x4E00 && code <= 0x9FFF) || (code >= 0x3400 && code <= 0x4DBF) ||
   506→   506→  1470→      (code >= 0x3000 && code <= 0x303F) || (code >= 0xFF00 && code <= 0xFFEF) ||
   507→   507→  1471→      (code >= 0x2E80 && code <= 0x2EFF);
   508→   508→  1472→    width += isCJK ? pt * 20 : pt * 11; // CJK: full-width, Latin: ~55% width
   509→   509→  1473→  }
   510→   510→  1474→  return width;
   511→   511→  1475→}
   512→   512→  1476→// Use estimateTextWidth() in calcTitleLayout() instead of simple char count
   513→   513→  1477→// to prevent mid-word breaks in mixed Chinese+English titles like "基于Transformer架构的..."
   514→   514→  1478→
   515→   515→  1479→// ── Meta info table ──
   516→   516→  1480→
   517→   517→  1481→// Calculate adaptive table and label column percentage based on longest label.
   518→   518→  1482→// Returns { tablePct, labelPct } — both as percentages.
   519→   519→  1483→// Uses ONLY WidthType.PERCENTAGE for cross-engine compatibility (MS Office + WPS).
   520→   520→  1484→function calcR5MetaLayout(metaEntries, fontPt = 12) {
   521→   521→  1485→  const maxLabelLen = Math.max(...metaEntries.map(e => [...e.label].length));
   522→   522→  1486→  // Label needs: maxLabelLen chars + "：" + 1 char padding
   523→   523→  1487→  const labelNeedTw = (maxLabelLen + 2) * fontPt * 20;
   524→   524→  1488→  // Value column: fixed ~5000tw for consistent underline length
   525→   525→  1489→  const valueNeedTw = 5000;
   526→   526→  1490→  const totalNeedTw = labelNeedTw + valueNeedTw;
   527→   527→  1491→  // Table width as % of page (11906tw), clamped to 55–75%
   528→   528→  1492→  const tablePct = Math.min(75, Math.max(55, Math.ceil(totalNeedTw / 11906 * 100)));
   529→   529→  1493→  // Label % within the table, clamped to 25–45%
   530→   530→  1494→  const rawLabelPct = Math.ceil(labelNeedTw / (tablePct / 100 * 11906) * 100);
   531→   531→  1495→  return { tablePct, labelPct: Math.max(25, Math.min(45, rawLabelPct)) };
   532→   532→  1496→}
   533→   533→  1497→
   534→   534→  1498→// Build R5 academic cover meta info table.
   535→   535→  1499→// ⚠️ CRITICAL cross-engine rules:
   536→   536→  1500→//   - Table width: WidthType.PERCENTAGE (NOT DXA — WPS breaks with DXA)
   537→   537→  1501→//   - Column widths: WidthType.PERCENTAGE
   538→   538→  1502→//   - Label column: LEFT aligned, plain text (NO full-width space padding)
   539→   539→  1503→//   - Value column: LEFT aligned, bottom border = fixed-length underline
   540→   540→  1504→//   - Table alignment: CENTER (visually centered on page)
   541→   541→  1505→function buildR5MetaTable(metaEntries) {
   542→   542→  1506→  // metaEntries: [{ label: "学院", value: "计算机科学与技术学院" }, ...]
   543→   543→  1507→  const { tablePct, labelPct } = calcR5MetaLayout(metaEntries);
   544→   544→  1508→  const valuePct = 100 - labelPct;
   545→   545→  1509→  const bottomBorder = { style: BorderStyle.SINGLE, size: 4, color: "000000" };
   546→   546→  1510→
   547→   547→  1511→  const rows = metaEntries.map(entry => new TableRow({
   548→   548→  1512→    children: [
   549→   549→  1513→      // Label cell: left-aligned, no bottom border
   550→   550→  1514→      new TableCell({
   551→   551→  1515→        width: { size: labelPct, type: WidthType.PERCENTAGE },
   552→   552→  1516→        borders: noBorders,
   553→   553→  1517→        margins: { top: 60, bottom: 60, left: 0, right: 0 },
   554→   554→  1518→        children: [new Paragraph({
   555→   555→  1519→          alignment: AlignmentType.LEFT,
   556→   556→  1520→          spacing: { before: 60, after: 60, line: 400 },
   557→   557→  1521→          children: [new TextRun({
   558→   558→  1522→            text: entry.label + "：",
   559→   559→  1523→            size: 24, font: { eastAsia: "SimSun", ascii: "Times New Roman" },
   560→   560→  1524→          })],
   561→   561→  1525→        })],
   562→   562→  1526→      }),
   563→   563→  1527→      // Value cell: left-aligned, bottom border = fixed-length underline
   564→   564→  1528→      new TableCell({
   565→   565→  1529→        width: { size: valuePct, type: WidthType.PERCENTAGE },
   566→   566→  1530→        borders: { top: NB, left: NB, right: NB, bottom: bottomBorder },
   567→   567→  1531→        margins: { top: 60, bottom: 60, left: 80, right: 0 },
   568→   568→  1532→        children: [new Paragraph({
   569→   569→  1533→          alignment: AlignmentType.LEFT,
   570→   570→  1534→          spacing: { before: 60, after: 60, line: 400 },
   571→   571→  1535→          children: [new TextRun({
   572→   572→  1536→            text: entry.value,
   573→   573→  1537→            size: 24, font: { eastAsia: "SimSun", ascii: "Times New Roman" },
   574→   574→  1538→          })],
   575→   575→  1539→        })],
   576→   576→  1540→      }),
   577→   577→  1541→    ],
   578→   578→  1542→  }));
   579→   579→  1543→
   580→   580→  1544→  return new Table({
   581→   581→  1545→    width: { size: tablePct, type: WidthType.PERCENTAGE },
   582→   582→  1546→    alignment: AlignmentType.CENTER,
   583→   583→  1547→    layout: TableLayoutType.FIXED,
   584→   584→  1548→    borders: allNoBorders,
   585→   585→  1549→    rows,
   586→   586→  1550→  });
   587→   587→  1551→}
   588→   588→  1552→
   589→   589→  1553→// ⚠️ MANDATORY: Cover section must use margin: 0. See "Cover Section Non-Negotiables" above.
   590→   590→  1554→function buildCoverR5(config) {
   591→   591→  1555→  const PAGE_H = 16838, SAFETY = 1200;
   592→   592→  1556→  const safeH = PAGE_H - SAFETY; // 15638
   593→   593→  1557→  const simMarginLR = 1701, simMarginT = 1200;
   594→   594→  1558→  const contentW = 11906 - simMarginLR * 2;
   595→   595→  1559→
   596→   596→  1560→  // ★ Width-aware title layout for mixed Chinese+English
   597→   597→  1561→  const { titlePt, titleLines } = calcTitleLayoutMixed(config.title, contentW, 36, 24);
   598→   598→  1562→  const titleSize = titlePt * 2;
   599→   599→  1563→
   600→   600→  1564→  // Parse meta entries
   601→   601→  1565→  const metaEntries = (config.metaLines || []).map(line => {
   602→   602→  1566→    const sep = line.indexOf("：") !== -1 ? "：" : ":";
   603→   603→  1567→    const idx = line.indexOf(sep);
   604→   604→  1568→    if (idx === -1) return { label: line, value: "" };
   605→   605→  1569→    return { label: line.slice(0, idx).trim(), value: line.slice(idx + sep.length).trim() };
   606→   606→  1570→  });
   607→   607→  1571→
   608→   608→  1572→  // Height budget (no margins.top — use spacing.before instead)
   609→   609→  1573→  const schoolNameH = config.schoolName ? (22 * 23 + 400) : 0;
   610→   610→  1574→  const titleTotalH = titleLines.length * (titlePt * 23 + 200);
   611→   611→  1575→  const subtitleH = config.subtitle ? (15 * 23 + 600) : 0;
   612→   612→  1576→  const metaRowH = 520; // 60+60 padding + ~400 line height
   613→   613→  1577→  const metaTableH = metaEntries.length * metaRowH;
   614→   614→  1578→  const footerH = config.footerRight ? (12 * 23 + 200) : 0;
   615→   615→  1579→  const spacerParas = 3 * 350;
   616→   616→  1580→  const fixedH = schoolNameH + titleTotalH + subtitleH + metaTableH + footerH + spacerParas;
   617→   617→  1581→  const remaining = Math.max(safeH - fixedH, 600);
   618→   618→  1582→
   619→   619→  1583→  // ★ topSpacing includes simulated top margin (simMarginT)
   620→   620→  1584→  const topSpacing = Math.min(Math.floor(remaining * 0.28) + simMarginT, 4200);
   621→   621→  1585→  const midSpacing = Math.min(Math.floor((remaining - simMarginT) * 0.18), 2000);
   622→   622→  1586→  const bottomSpacing = Math.min(remaining - topSpacing + simMarginT - midSpacing, 5500);
   623→   623→  1587→
   624→   624→  1588→  const children = [];
   625→   625→  1589→  children.push(new Paragraph({ spacing: { before: topSpacing } }));
   626→   626→  1590→
   627→   627→  1591→  // School name (optional)
   628→   628→  1592→  if (config.schoolName) {
   629→   629→  1593→    children.push(new Paragraph({
   630→   630→  1594→      alignment: AlignmentType.CENTER, spacing: { after: 400 },
   631→   631→  1595→      children: [new TextRun({ text: config.schoolName, size: 44, characterSpacing: 40,
   632→   632→  1596→        font: { eastAsia: "SimSun", ascii: "Times New Roman" } })],
   633→   633→  1597→    }));
   634→   634→  1598→  }
   635→   635→  1599→
   636→   636→  1600→  // Title
   637→   637→  1601→  for (let i = 0; i < titleLines.length; i++) {
   638→   638→  1602→    children.push(new Paragraph({
   639→   639→  1603→      alignment: AlignmentType.CENTER, spacing: { after: i < titleLines.length - 1 ? 120 : 300 },
   640→   640→  1604→      children: [new TextRun({ text: titleLines[i], size: titleSize, bold: true,
   641→   641→  1605→        font: { eastAsia: "SimHei", ascii: "Times New Roman" } })],
   642→   642→  1606→    }));
   643→   643→  1607→  }
   644→   644→  1608→
   645→   645→  1609→  // Subtitle
   646→   646→  1610→  if (config.subtitle) {
   647→   647→  1611→    children.push(new Paragraph({
   648→   648→  1612→      alignment: AlignmentType.CENTER, spacing: { after: 200 },
   649→   649→  1613→      children: [new TextRun({ text: config.subtitle, size: 30,
   650→   650→  1614→        font: { eastAsia: "SimSun", ascii: "Times New Roman" } })],
   651→   651→  1615→    }));
   652→   652→  1616→  }
   653→   653→  1617→
   654→   654→  1618→  children.push(new Paragraph({ spacing: { before: midSpacing } }));
   655→   655→  1619→
   656→   656→  1620→  // Meta info table
   657→   657→  1621→  if (metaEntries.length > 0) children.push(buildR5MetaTable(metaEntries));
   658→   658→  1622→
   659→   659→  1623→  children.push(new Paragraph({ spacing: { before: bottomSpacing } }));
   660→   660→  1624→
   661→   661→  1625→  // Footer
   662→   662→  1626→  if (config.footerRight) {
   663→   663→  1627→    children.push(new Paragraph({

   665→   665→  1629→      children: [new TextRun({ text: config.footerRight, size: 24, color: "404040",
   666→   666→  1630→        font: { eastAsia: "SimSun", ascii: "Times New Roman" } })],
   667→   667→  1631→    }));
   668→   668→  1632→  }
   669→   669→  1633→
   670→   670→  1634→  // ★ 16838 outer wrapper — only left/right margins, NO margins.top
   671→   671→  1635→  return [new Table({

   688→   688→  1652→}

   696→   696→  1660→function blendColors(hex1, hex2, ratio) {
   697→   697→  1661→  const p = (s, i) => parseInt(s.replace("#","").slice(i, i+2), 16);
   698→   698→  1662→  const mix = (c1, c2) => Math.round(c1 + (c2 - c1) * ratio);
   699→   699→  1663→  const r = mix(p(hex1,0), p(hex2,0)), g = mix(p(hex1,2), p(hex2,2)), b = mix(p(hex1,4), p(hex2,4));
   700→   700→  1664→  return [r, g, b].map(v => v.toString(16).padStart(2,"0")).join("");
   701→   701→  1665→}

   753→   753→  1717→**Color usage rules:**
   754→   754→  1718→- Max **5 colors** per chart; excess categories use depth variants of same hue
   755→   755→  1719→- Emphasis data uses document accent color; non-emphasis uses grey `#B0B0B0`
   756→   756→  1720→- Adjacent segments in pie/bar charts must have hue gap ≥ 60°
   757→   757→  1721→
   758→   758→  1722→1. **Anti-overlap**: If >6 x-axis labels, rotate 45° (`plt.xticks(rotation=45, ha='right')`)
   759→   759→  1723→2. **Anti-stretch**: Always set figure size explicitly (`fig, ax = plt.subplots(figsize=(10, 6))`)
   760→   760→  1724→3. **Aspect ratio (CRITICAL)**: When embedding in docx, MUST read actual image dimensions and calculate height proportionally. NEVER hardcode both width and height — pie charts become ellipses, radar charts become diamonds.
   761→   761→  1725→   ```js
   762→   762→  1726→   const sizeOf = require("image-size");
   763→   763→  1727→   const dims = sizeOf(chartBuffer);
   764→   764→  1728→   const displayWidth = 500;
   765→   765→  1729→   const displayHeight = Math.round(displayWidth * (dims.height / dims.width));
   766→   766→  1730→   // transformation: { width: displayWidth, height: displayHeight }
   767→   767→  1731→   ```
   768→   768→  1732→4. **DPI**: Save at 200+ DPI (`plt.savefig("chart.png", dpi=200, bbox_inches="tight")`)
   769→   769→  1733→5. **Colors**: Use palette accent color for primary data series
   770→   770→  1734→6. **Legend**: Place outside plot area if >4 series
   771→   771→  1735→7. **Square charts**: Pie and radar charts MUST use `figsize=(8, 8)` (equal width/height) to preserve circular/radial shape
   772→   772→  1736→7. **Grid**: Light gray grid (`ax.grid(True, alpha=0.3)`)
